# Join by country code and keep all economies in the accident data.
combined <- merge(
  accidents,
  gdp[c("country_code", "mean_usd")],
  by.x = "Country Code",
  by.y = "country_code",
  all.x = TRUE
)

# Keep rows with a positive GDP and an available accident rate.
plot_data <- subset(
  combined,
  is.finite(mean_usd) & mean_usd > 0 & is.finite(Accidents_per_million)
)

# Plot one point per economy.
plot(
  plot_data$mean_usd / 1e9,
  plot_data$Accidents_per_million,
  log = "x",
  pch = 19,
  col = "steelblue",
  xlab = "Average GDP (current US$ billions, log scale)",
  ylab = "Accidents per million departures"
)
# Measure the rank correlation between GDP and the accident rate.
rho <- cor(
  plot_data$mean_usd,
  plot_data$Accidents_per_million,
  method = "spearman"
)

nrow(plot_data)
## [1] 174
round(rho, 3)
## [1] 0.231

Results

Across the 174 economies with usable data, average GDP and recorded aviation accidents per million departures had a weak positive rank correlation (Spearman’s rho = 0.231). This describes an association across economies and does not establish a causal effect of GDP.

Higher vs. lower GDP comparison

I split the matched economies at the median of their 2002–2023 average GDP. Economies at or above the median are in the higher GDP group. For each group, I divided total accidents by total departures and multiplied by one million.

cutoff <- median(plot_data$mean_usd)
plot_data$GDP_group <- ifelse(plot_data$mean_usd < cutoff,
                             "Lower GDP", "Higher GDP")

totals <- aggregate(cbind(Accidents, Departures) ~ GDP_group,
                    data = plot_data, FUN = sum)
totals$Economies <- as.integer(table(plot_data$GDP_group)[totals$GDP_group])
totals$Accidents_per_million <- round(totals$Accidents / totals$Departures * 1e6, 3)
